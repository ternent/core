import assert from "node:assert/strict";
import { readFileSync } from "node:fs";
import { identityFromSeed, generateIdentity, encryptAge, decryptAgeWithSecret } from "@ternent/armour";
import { validatePublicIdentity } from "@ternent/identity";
import { createDocument, openDocument } from "@ternent/concord";
import { createStandardSystemProfile, setMember, removeMember } from "@ternent/concord/system";
import { registerModel, createMemoryArtifactStore, assertModelConformance } from "@ternent/concord/testing";
import { canonicalize, hashEntrySubject, hashCommitSubject, verifyDocument, compareSnapshots } from "@ternent/concord/protocol";
const value = result => { assert.equal(result.ok, true, result.error?.code); return result.value; };
const fixed = JSON.parse(readFileSync("owner.expected.json", "utf8"));
const identity = value(await identityFromSeed(new Uint8Array(Buffer.from(fixed.identitySeedHex,"hex"))));
assert.equal(validatePublicIdentity(identity.publicIdentity).ok, true);
const system = createStandardSystemProfile();
const handle = value(await createDocument({model:registerModel,system,identity}));
value(await handle.dispatch("set",{value:"packed Concord"}));
const artifact = handle.export();
const opened = await openDocument({bytes:artifact,model:registerModel,system,signer:identity});
assert.equal(opened.state().status,"ready");
assert.deepEqual(value(await opened.replay()),{value:"packed Concord"});
assert.deepEqual(opened.inspect().findings,[]);
assert.equal((await opened.verify()).integrity,"valid");
assert.equal((await verifyDocument(artifact)).integrity,"valid");
assert.equal(value(await compareSnapshots(artifact,opened.export())),"equal");
const saved = new Uint8Array(readFileSync("owner.document.json"));
const reference = await openDocument({bytes:saved,model:registerModel,system,signer:identity});
assert.deepEqual(reference.state(),{status:"ready",documentId:fixed.documentId,head:fixed.head,projection:fixed.projection,canWrite:true});
const wire = JSON.parse(new TextDecoder().decode(saved));
assert.equal(value(hashEntrySubject(wire.entries[0].subject)),wire.entries[0].id);
assert.equal(value(hashCommitSubject(wire.commits[0].subject)),wire.commits[0].id);
assert.equal(new TextDecoder().decode(value(canonicalize({z:1,a:2}))),'{"a":2,"z":1}');
const locked = await openDocument({bytes:saved,model:registerModel,system});
assert.equal(locked.state().reason,"locked");
assert.deepEqual(locked.export(),saved);
const store = createMemoryArtifactStore();value(await store.write(artifact));assert.deepEqual(value(await store.read()),artifact);
assert.equal(assertModelConformance(registerModel,{initial:{value:""},steps:[]}).ok,true);
const fresh = value(await generateIdentity());assert.notEqual(fresh.publicIdentity.id,identity.publicIdentity.id);
const membership = value(await createDocument({model:registerModel,system:createStandardSystemProfile(),identity}));
value(await membership.dispatch("set",{value:"history before admission"}));
value(await setMember(membership,{identity:fresh.publicIdentity,capabilities:["read"]}));
const viewer = await openDocument({bytes:membership.export(),model:registerModel,system:createStandardSystemProfile(),reader:fresh});
assert.deepEqual(value(await viewer.replay()),{value:"history before admission"});
assert.equal(viewer.state().canWrite,false);
assert.equal(viewer.inspect().otherRecipientsDecryptability,"unverified");
value(await removeMember(membership,{identityId:fresh.publicIdentity.id}));
const revoked = await openDocument({bytes:membership.export(),model:registerModel,system:createStandardSystemProfile(),reader:fresh});
assert.equal(revoked.inspect().findings[0].code,"READ_ACCESS_REQUIRED");
const group = new Uint8Array(Buffer.from(fixed.groupSecretHex,"hex"));
const cipher = value(await encryptAge({recipients:[fixed.groupRecipient],plaintext:new TextEncoder().encode("standalone Armour")}));
assert.equal(new TextDecoder().decode(value(await decryptAgeWithSecret({secret:group,ciphertext:cipher}))),"standalone Armour");

const annotated = structuredClone(wire);
annotated.metadata = { n: 1000000000000000 };
const canonical = value(canonicalize(annotated));
const transport = new TextEncoder().encode(new TextDecoder().decode(canonical).replace("1000000000000000", "1e15"));
const bounded = await openDocument({bytes:transport,model:registerModel,system,signer:identity,limits:{maxBytes:transport.length}});
assert.deepEqual(bounded.state(),{status:"inspection",reason:"incompatible"});
assert.equal((await bounded.verify()).findings[0].code,"RESOURCE_LIMIT");
assert.deepEqual(bounded.export(),transport);
const expanded = await openDocument({bytes:transport,model:registerModel,system,signer:identity,limits:{maxBytes:canonical.length}});
assert.equal(expanded.state().status,"ready");
assert.deepEqual(expanded.export(),canonical);
assert.equal((await expanded.verify()).integrity,"valid");

const tampered = structuredClone(wire);
tampered.commits[1].signature = "A".repeat(86);
const badBytes = value(canonicalize(tampered));
const inspection = await openDocument({bytes:badBytes,model:registerModel,system,signer:identity});
assert.deepEqual(inspection.state(),{status:"inspection",reason:"invalid"});
assert.equal(inspection.inspect().findings[0].code,"SIGNATURE_INVALID");
assert.deepEqual(inspection.inspect().history,tampered.commits);
assert.deepEqual(inspection.export(),badBytes);
assert.equal((await inspection.dispatch("set",{value:"blocked"})).error.code,"INSPECTION_ONLY");

const multi = {...registerModel,commands:{set:{...registerModel.commands.set,decide:()=>({ok:true,value:Array.from({length:20},()=>({type:"value.set",version:1,data:{value:""}}))})}}};
const batch = value(await createDocument({model:multi,system,identity}));
value(await batch.dispatch("set",{value:""}));
const batchBytes = batch.export();
const batchOpen = await openDocument({bytes:batchBytes,model:multi,system:createStandardSystemProfile(),signer:identity,limits:{maxBytes:batchBytes.length}});
assert.deepEqual(batchOpen.state(),batch.state());
assert.deepEqual(batchOpen.export(),batchBytes);
assert.deepEqual(value(await batchOpen.replay()),{value:""});
assert.equal((await batchOpen.verify()).integrity,"valid");
console.log(`Concord public consumers passed on ${process.version}`);
