import assert from "node:assert/strict";
import { identityFromSeed } from "@ternent/armour";
import { createDocument, openDocument } from "@ternent/concord";
import { createStandardSystemProfile } from "@ternent/concord/system";
import { registerModel, createMemoryArtifactStore, createMemoryExpectedHeadStore } from "@ternent/concord/testing";
import { compareSnapshots, verifyDocument } from "@ternent/concord/protocol";
const value = result => { assert.equal(result.ok,true,result.error?.code);return result.value; };
const identity = value(await identityFromSeed(new Uint8Array(32).fill(7)));
const create = () => createDocument({model:registerModel,system:createStandardSystemProfile(),identity});
const open = (bytes,extra={}) => openDocument({bytes,model:registerModel,system:createStandardSystemProfile(),signer:identity,...extra});
const local = value(await create());
const base = local.export();
const baseHead = local.state().head;
const store = createMemoryExpectedHeadStore();
assert.equal(store.expectedHead,true);
assert.equal(value(await store.read()),null);
value(await store.writeExpected(base,null));
assert.equal((await store.writeExpected(base,null)).error.code,"HEAD_CONFLICT");
const left = await open(base), right = await open(base);
value(await left.dispatch("set",{value:"left branch"}));
value(await right.dispatch("set",{value:"right branch"}));
const localSnapshots = [left.export(),right.export()];
const results = await Promise.all(localSnapshots.map(bytes=>store.writeExpected(bytes,baseHead)));
assert.equal(results.filter(result=>result.ok).length,1);
const loser = results.findIndex(result=>!result.ok);
const conflict = results[loser].error;
assert.equal(conflict.code,"HEAD_CONFLICT");
assert(conflict.remote instanceof Uint8Array);
assert.equal(value(await compareSnapshots(localSnapshots[loser],conflict.remote)),"diverged");
assert.deepEqual(value(await store.read()),conflict.remote);
assert.deepEqual((loser === 0 ? left : right).export(),localSnapshots[loser]);
const storedCopy = new Uint8Array(conflict.remote);
conflict.remote.fill(0);
assert.deepEqual(value(await store.read()),storedCopy);
assert.equal((await verifyDocument(localSnapshots[loser])).integrity,"valid");
const published = await open(value(await store.read()));
assert.equal(published.state().status,"ready");
const anchored = await open(base,{anchor:storedCopy});
assert.equal(anchored.state().reason,"freshness");
assert.equal(anchored.inspect().findings[0].code,"STALE_HEAD");
assert.equal((await open(base)).state().status,"ready");

const draft = await open(storedCopy,{mode:"draft"});
value(await draft.dispatch("set",{value:"unpublished draft"}));
const pending = draft.draft();
const committed = draft.export();
const failedStore = {async write(){return {ok:false,error:{code:"STORAGE_FAILED",phase:"storage",message:"Injected I/O failure"}}}};
assert.equal((await failedStore.write(draft.export())).error.code,"STORAGE_FAILED");
assert.deepEqual(draft.export(),committed);
assert.deepEqual(draft.draft(),pending);
const raw = createMemoryArtifactStore();
value(await raw.write(draft.export()));
const reopened = await open(value(await raw.read()),{mode:"draft"});
assert.deepEqual(reopened.state(),draft.state());
assert.equal(reopened.draft(),null);
assert.notEqual(reopened.state().projection.value,"unpublished draft");
const input = new Uint8Array([1,2,3]);
value(await raw.write(input));input.fill(0);
const read = value(await raw.read());assert.deepEqual(read,new Uint8Array([1,2,3]));read.fill(0);
assert.deepEqual(value(await raw.read()),new Uint8Array([1,2,3]));
assert.equal((await store.writeExpected(new Uint8Array([1]),published.state().head)).error.code,"STORAGE_FAILED");
assert.deepEqual(value(await store.read()),storedCopy);
console.log(`External storage, atomic expected-head conflicts, retained forks, anchors and draft isolation passed on ${process.version}`);
