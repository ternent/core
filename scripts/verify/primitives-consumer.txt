import assert from "node:assert/strict";
import {
  createIdentityFromMnemonic,
  createMnemonicIdentity,
  decodeBase64Url,
  deriveAgeRecipient,
  deriveKeyId,
  getDefaultSignatureContext,
  parseIdentity,
  serializeIdentity,
  signerFromSeed,
  signUtf8,
  validateIdentity,
  verifySignature,
  verifyUtf8,
} from "@ternent/identity";
import {
  encryptForIdentities,
  encryptForRecipients,
  decryptWithIdentity,
  withAgeDecryption,
} from "@ternent/armour";

const value = (result) => {
  assert.equal(result.ok, true, result.error?.code);
  return result.value;
};
const createdAt = "2026-11-20T00:00:00.000Z";
const { identity, mnemonic } = await createMnemonicIdentity({ words: 12, createdAt });
const restored = await createIdentityFromMnemonic({ mnemonic, createdAt });
assert.deepEqual(restored, identity);
assert.deepEqual(parseIdentity(serializeIdentity(identity)), identity);
assert.deepEqual(await validateIdentity(identity), identity);
assert.equal(await deriveKeyId(identity.publicKey), identity.keyId);

const signer = value(await signerFromSeed(value(decodeBase64Url(identity.material.seed))));
assert.equal(signer.publicIdentity.publicKey, identity.publicKey);
assert.equal(signer.publicIdentity.id, `identity:${identity.keyId}`);
const plaintext = new TextEncoder().encode("installed primitive contracts");
const rawSignature = value(await signer.sign(plaintext));
assert.equal(value(await verifySignature(signer.publicIdentity, plaintext, rawSignature)), true);
const signature = await signUtf8(identity, new TextDecoder().decode(plaintext));
assert.equal(getDefaultSignatureContext(), "ternent-seal/v2");
assert.equal(await verifyUtf8(identity.publicKey, new TextDecoder().decode(plaintext), signature), true);
assert.equal(
  await verifyUtf8(identity.publicKey, new TextDecoder().decode(plaintext), signature, { context: "other-context" }),
  false,
);
assert.equal((await verifySignature(signer.publicIdentity, plaintext, value(decodeBase64Url(signature)))).ok, false);

const recipient = await deriveAgeRecipient(identity);
const ciphertext = await encryptForRecipients({ recipients: [recipient], data: plaintext });
assert.match(new TextDecoder().decode(ciphertext), /^-----BEGIN AGE ENCRYPTED FILE-----/);
assert.deepEqual(await decryptWithIdentity({ identity, data: ciphertext }), plaintext);
const binary = await encryptForIdentities({ identities: [identity], data: plaintext, output: "binary" });
assert.deepEqual(value(await withAgeDecryption(signer).decryptAge(binary)), plaintext);
const { identity: stranger } = await createMnemonicIdentity();
await assert.rejects(decryptWithIdentity({ identity: stranger, data: ciphertext }), { code: "ARMOUR_DECRYPT_FAILED" });

console.log(`Identity host helpers, exact/contextual signing and init-free Armour passed on ${process.version}`);
