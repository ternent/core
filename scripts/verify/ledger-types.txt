import { createLedger, encodePayload, decodePayload, type LedgerPayloadCodec, type LedgerPayloadEncoder, type LedgerPayloadDecoder, type LedgerReplayEntry } from "@ternent/ledger";
import { createAgeCodec, generateIdentity, type AgePayloadEncoder, type AgePayloadDecoder } from "@ternent/armour";
import { ageRecipientFromPublicIdentity, generateSigner } from "@ternent/identity";
const signer = await generateSigner();
if (!signer.ok) throw new Error(signer.error.code);
const log = await createLedger({signer: signer.value});
await log.create();
await log.append({kind:"reading",payload:{value:7}});
const entries: LedgerReplayEntry[] = await log.replay();
const projected = await createLedger({signer:signer.value,initialProjection:0,projector:(count,entry)=>count+1});
const count: number = await projected.replay();
// @ts-expect-error A kind is required.
await log.append({payload:42});
// @ts-expect-error Payloads are portable JSON.
await log.append({kind:"bad",payload:1n});
// @ts-expect-error A custom projection requires its projector.
await createLedger({initialProjection:0});
// @ts-expect-error The selected projection type survives package declarations.
const wrong: string = await projected.replay();
void [entries,count,wrong];
const reader = await generateIdentity(); if (!reader.ok) throw new Error(reader.error.code);
const codec: LedgerPayloadCodec = createAgeCodec({reader:reader.value});
const recipient = ageRecipientFromPublicIdentity(reader.value.publicIdentity); if (!recipient.ok) throw new Error(recipient.error.code);
const recipients = [recipient.value];
const writer = createAgeCodec({recipients});
const readerOnly = createAgeCodec({reader:reader.value});
const both = createAgeCodec({recipients,reader:reader.value});
const encoder: LedgerPayloadEncoder = writer;
const decoder: LedgerPayloadDecoder = readerOnly;
const armourEncoder: AgePayloadEncoder = writer;
const armourDecoder: AgePayloadDecoder = readerOnly;
await writer.encode(new Uint8Array());
await readerOnly.decode(new Uint8Array());
await both.encode(new Uint8Array());
await both.decode(new Uint8Array());
await encodePayload({value:7},writer);
await encodePayload({value:7},both);
// @ts-expect-error A reader-only codec cannot encode payloads.
await encodePayload({value:7},readerOnly);
// @ts-expect-error A codec with no operations cannot encode payloads.
await encodePayload({value:7},createAgeCodec());
// @ts-expect-error An optional encoder is not a guaranteed encoding capability.
await encodePayload({value:7},codec);
const dynamicOptions: {recipients?: readonly string[];reader?: typeof reader.value} = {};
const dynamicCodec = createAgeCodec(dynamicOptions);
// @ts-expect-error Optional configuration cannot promise an encoder.
await encodePayload({value:7},dynamicCodec);
const decoded = await decodePayload((await log.replay())[0].payload.data,codec);
if (decoded.status === "locked") {
  // @ts-expect-error Unavailable payloads do not expose plaintext.
  decoded.data;
}
// @ts-expect-error Codec integration retains the portable JSON requirement.
await encodePayload(1n,writer);
// @ts-expect-error Byte codecs cannot return text instead of bytes.
const invalidCodec: LedgerPayloadCodec = {id:"broken",encode:async()=>"text"};
void [invalidCodec,encoder,decoder,armourEncoder,armourDecoder];
