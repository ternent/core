import assert from "node:assert/strict";
import { readFileSync, writeFileSync, mkdirSync, statSync, rmSync } from "node:fs";
import { resolve } from "node:path";
import { spawnSync } from "node:child_process";
import { createSealIdentity, exportIdentityJson } from "@ternent/seal";
import { createSealProof, getSealProofSigningPayload, verifySealProofAgainstBytes, verifySealProofSignature, createSealHash } from "@ternent/seal/proof";
import { createSealArtifact, verifySealArtifact, decryptSealArtifactPayload, parseSealArtifactJson } from "@ternent/seal/artifact";
import { stringifySealManifest, parseSealManifestJson } from "@ternent/seal/manifest";
import { verifyPublicKeyKeyId, verifySealUtf8 } from "@ternent/seal/crypto";
import { EXIT_KEY_CONFIG, EXIT_INVALID_PROOF, EXIT_SIGNATURE_INVALID, EXIT_HASH_MISMATCH } from "@ternent/seal/errors";
import { recipientFromIdentity } from "@ternent/armour";
const fixed = JSON.parse(readFileSync("seal-proof.v2.json","utf8"));
const content = new TextEncoder().encode(fixed.contentUtf8);
assert.equal(await createSealHash(content),fixed.subjectHash);
assert.equal(getSealProofSigningPayload(fixed.proof),fixed.signingPayload);
assert.deepEqual(await createSealProof({signer:{identity:fixed.identity},subject:fixed.proof.subject,createdAt:fixed.proof.createdAt}),fixed.proof);
assert.equal((await verifySealProofAgainstBytes(fixed.proof,content)).valid,true);
assert.equal(await verifySealUtf8(fixed.proof.signature,fixed.signingPayload,fixed.proof.signer.publicKey),true);
assert.equal(await verifyPublicKeyKeyId(fixed.proof.signer.publicKey,fixed.proof.signer.keyId),true);
assert.equal((await verifySealProofSignature({...fixed.proof,signer:{...fixed.proof.signer,publicKey:"malformed"}})).ok,false);
assert.equal(stringifySealManifest(fixed.manifest),fixed.canonicalManifest);
assert.deepEqual(parseSealManifestJson(fixed.canonicalManifest).manifest,fixed.manifest);
assert.equal(Object.hasOwn(parseSealManifestJson(fixed.canonicalManifest).manifest.files,"__proto__"),true);
const recipient = await createSealIdentity();
const outsider = await createSealIdentity();
const encrypted = await createSealArtifact({signer:{identity:fixed.identity},payload:content,subjectPath:"test.txt",recipients:[await recipientFromIdentity(recipient)]});
assert.equal((await verifySealArtifact(encrypted)).valid,true);
assert.deepEqual(await decryptSealArtifactPayload({artifact:encrypted,identity:recipient}),content);
await assert.rejects(()=>decryptSealArtifactPayload({artifact:encrypted,identity:outsider}),error=>error.code === "SEAL_DECRYPTION_FAILED");
const altered = structuredClone(encrypted); altered.payload.data += "tamper";
assert.equal((await verifySealArtifact(altered)).valid,false);
await assert.rejects(()=>decryptSealArtifactPayload({artifact:altered,identity:recipient}));

const bin = resolve("node_modules/@ternent/seal/bin/seal");
assert.equal(readFileSync(bin,"utf8").startsWith("#!/usr/bin/env node\n"),true);
assert.notEqual(statSync(bin).mode & 0o111,0);
const keyPath = resolve("seal-identity.json");
const env = {...process.env,SEAL_IDENTITY:"",SEAL_IDENTITY_FILE:keyPath};
function cli(args,status=0,config=env) {
  const result = spawnSync(process.execPath,[bin,...args],{env:config,encoding:"utf8"});
  assert.equal(result.status,status,`seal ${args[0]}: ${result.stderr}`);
  return result.stdout;
}
assert.match(cli(["--help"],0,{...env,SEAL_IDENTITY_FILE:""}),/Usage: seal/);
cli(["identity","create","--out",keyPath,"--words","12","--passphrase=with=equals","--mnemonic-out","seal-backup.txt","--quiet"]);
assert.equal(statSync(keyPath).mode & 0o777,0o600);
assert.equal(statSync("seal-backup.txt").mode & 0o777,0o600);
assert.match(readFileSync("seal-backup.txt","utf8"),/Passphrase:\nwith=equals/);
const createdIdentity = JSON.parse(readFileSync(keyPath,"utf8"));
assert.equal(createdIdentity.version,"2");
rmSync("seal-assets",{recursive:true,force:true});
mkdirSync("seal-assets/nested",{recursive:true});
writeFileSync("seal-assets/index.html","<main>Seal packed candidate</main>");
writeFileSync("seal-assets/nested/example.txt","signed bytes");
writeFileSync("seal-assets/__proto__","own filename");
// The inspected Action's configurable CLI runs this same sequence on a clean build directory.
cli(["manifest","create","--input","seal-assets","--out","seal-assets/dist-manifest.json","--quiet"]);
cli(["sign","--input","seal-assets/dist-manifest.json","--out","seal-assets/proof.json","--quiet"]);
writeFileSync("seal-assets/public-key.json",cli(["public-key","--json"]));
const manifest = JSON.parse(readFileSync("seal-assets/dist-manifest.json","utf8"));
assert.equal(Object.hasOwn(manifest.files,"__proto__"),true);
const proof = JSON.parse(readFileSync("seal-assets/proof.json","utf8"));
const publicKey = JSON.parse(readFileSync("seal-assets/public-key.json","utf8"));
assert.equal(proof.subject.kind,"manifest");
assert.equal(publicKey.publicKey,proof.signer.publicKey);
assert.equal(publicKey.keyId,proof.signer.keyId);
assert.equal(JSON.parse(cli(["verify","--proof","seal-assets/proof.json","--input","seal-assets/dist-manifest.json","--json"])).valid,true);
const manifestBytes = new Uint8Array(readFileSync("seal-assets/dist-manifest.json"));
assert.equal((await verifySealProofAgainstBytes(proof,manifestBytes)).valid,true);
writeFileSync("different.txt","different bytes");
cli(["verify","--proof","seal-assets/proof.json","--input","different.txt","--quiet"],EXIT_HASH_MISMATCH);
const badSignature = structuredClone(proof);badSignature.signature="A".repeat(86);
writeFileSync("bad-signature.json",JSON.stringify(badSignature));
cli(["verify","--proof","bad-signature.json","--input","seal-assets/dist-manifest.json","--quiet"],EXIT_SIGNATURE_INVALID);
writeFileSync("invalid-proof.json",'{"type":"bad"}');
cli(["verify","--proof","invalid-proof.json","--input","different.txt","--quiet"],EXIT_INVALID_PROOF);
cli(["public-key","--quiet"],EXIT_KEY_CONFIG,{...env,SEAL_IDENTITY_FILE:""});
cli(["manifest","create","--quiet"],1);
cli(["public-key","--quiet"],EXIT_KEY_CONFIG,{...env,SEAL_IDENTITY:"not-json"});
const mismatchedIdentity = {...fixed.identity,material:outsider.material};
cli(["public-key","--quiet"],EXIT_KEY_CONFIG,{...env,SEAL_IDENTITY:JSON.stringify(mismatchedIdentity)});
const ageRecipient = await recipientFromIdentity(recipient);
cli(["sign","--input","different.txt","--recipient",ageRecipient,"--out","encrypted-artifact.json","--quiet"]);
assert.equal(JSON.parse(cli(["verify","--artifact","encrypted-artifact.json","--json"])).valid,true);
const parsedArtifact = parseSealArtifactJson(readFileSync("encrypted-artifact.json","utf8"));
assert.equal(parsedArtifact.ok,true);
assert.equal(new TextDecoder().decode(await decryptSealArtifactPayload({artifact:parsedArtifact.artifact,identity:recipient})),"different bytes");
writeFileSync("invalid-artifact.json",'{"type":"bad"}');
cli(["verify","--artifact","invalid-artifact.json","--quiet"],EXIT_INVALID_PROOF);
assert.match(exportIdentityJson(createdIdentity),/ternent-identity/);
console.log(`Seal public paths, fixed proof, encrypted artifacts, executable, Action command sequence and exits passed on ${process.version}`);
