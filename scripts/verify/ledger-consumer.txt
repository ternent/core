import assert from "node:assert/strict";
import { isBuiltin, registerHooks } from "node:module";
import { readFileSync } from "node:fs";
const fixed = new Uint8Array(readFileSync("ledger-signed-log.json"));
registerHooks({resolve(specifier,context,next){
  if (isBuiltin(specifier) || specifier.startsWith("@ternent/concord") || specifier.startsWith("@ternent/armour") || specifier.startsWith("@ternent/seal") || specifier.startsWith("age-encryption")) throw new Error(`Forbidden Ledger dependency: ${specifier}`);
  return next(specifier,context);
}});
const {createLedger,verifyLedger,compareLedgers,encodeJson}=await import("@ternent/ledger");
await import("@ternent/ledger/protocol");
const {signerFromSeed}=await import("@ternent/identity");
const identity=await signerFromSeed(new Uint8Array(32).fill(4));if(!identity.ok)throw new Error(identity.error.code);
assert.equal((await verifyLedger(fixed)).integrity,"valid");
const log=await createLedger({signer:identity.value});
await log.load(fixed);
const base=await log.export();
await log.append({kind:"standalone",payload:{value:42}});
assert.deepEqual(await log.export(),base);
assert.equal((await log.replay()).at(-1).committed,false);
const receipt=await log.commit({metadata:{note:"packed consumer"}});
assert.equal(receipt.committedEntries.length,1);
const wire=await log.export();
assert.equal((await log.verify()).integrity,"valid");
assert.deepEqual(await compareLedgers(fixed,encodeJson(wire)),{ok:true,value:"behind"});
const reader=await createLedger();await reader.import(wire);
assert.deepEqual(await reader.replay(),await log.replay());
const tampered=structuredClone(wire);tampered.entries.at(-1).subject.payload.data={value:0};
await assert.rejects(()=>reader.import(tampered));
assert.deepEqual(await reader.export(),wire);
console.log("Standalone Ledger create/load/stage/commit/replay/verify and fixed vector pass with Node builtins, Concord, Armour and Seal imports blocked");
