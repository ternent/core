import assert from "node:assert/strict";
import { ageRecipientFromPublicIdentity } from "@ternent/identity";
import { createAgeCodec, identityFromSeed } from "@ternent/armour";
import { createLedger, encodePayload, decodePayload } from "@ternent/ledger";
const value = result => { assert.equal(result.ok,true,result.error?.code); return result.value; };
const alice = value(await identityFromSeed(new Uint8Array(32).fill(11)));
const bob = value(await identityFromSeed(new Uint8Array(32).fill(12)));
const codec = createAgeCodec({recipients:[value(ageRecipientFromPublicIdentity(alice.publicIdentity))],reader:alice});
const log = await createLedger({signer:alice}); await log.create();
await log.append({kind:"note",payload:await encodePayload({text:"private packed payload"},codec)});
assert.equal((await log.export()).entries.length,0);
await log.commit();
const wire = await log.export();
assert.equal(JSON.stringify(wire).includes("private packed payload"),false);
const reader = await createLedger(); await reader.load(wire);
assert.equal((await reader.verify()).integrity,"valid");
const payload = (await reader.replay())[0].payload.data;
assert.deepEqual(await decodePayload(payload),{status:"locked",codec:"age"});
assert.deepEqual(await decodePayload(payload,createAgeCodec({reader:bob})),{status:"locked",codec:"age"});
assert.deepEqual(await decodePayload(payload,codec),{status:"decoded",data:{text:"private packed payload"}});
assert.deepEqual(await reader.export(),wire);
const tampered = structuredClone(wire);tampered.entries[0].subject.payload.data.data += "A";
await assert.rejects(()=>reader.load(tampered));
console.log(`Opt-in Armour codec with ordinary Ledger staging, keyless verification, locked/readable payloads and tampering passed on ${process.version}`);
