import { createIdentity, verify } from "@ternent/identity";
import { createSealHash, createSealProof, verifySealProof } from "@ternent/seal";
import { decrypt, encrypt } from "@ternent/armour";
import { createLedger, createWriter, openLedger } from "@ternent/ledger";
import { createDocument, defineModel, openDocument } from "@ternent/concord";

type Notes = { notes: { [id: string]: string } };

const notes = defineModel<Notes>({
  id: "notes",
  version: 1,
  initial: () => ({ notes: {} }),
  reduce(state, event) {
    const data = event.data as { id: string; title: string };
    if (event.type === "note.put") return { notes: { ...state.notes, [data.id]: data.title } };
    throw new Error(`Unknown event ${event.type}`);
  },
});

function check(condition: boolean, message: string): void {
  if (!condition) throw new Error(`browser ${message} failed`);
}

// Runs every library in the browser through its installed, built package.
export async function proveConcord(): Promise<void> {
  const text = new TextEncoder();
  const alice = await createIdentity();
  const bob = await createIdentity();
  const signature = await alice.sign(text.encode("browser"));
  check(await verify(alice.publicIdentity, text.encode("browser"), signature), "Identity");

  const proof = await createSealProof({
    signer: alice,
    subject: { hash: await createSealHash(text.encode("browser")) },
  });
  check((await verifySealProof(proof, text.encode("browser"))).valid, "Seal");

  const ciphertext = await encrypt(text.encode("browser"), [bob.publicIdentity]);
  check(new TextDecoder().decode(await decrypt(ciphertext, bob)) === "browser", "Armour");

  const writer = createWriter(await createLedger({ signer: alice }), alice);
  writer.append({ kind: "browser", payload: { value: "signed" } });
  const history = await writer.commit();
  check((await openLedger(history.export())).compare(history) === "equal", "Ledger");

  const doc = await createDocument({ model: notes, identity: alice });
  const team = await doc.createGroup();
  await doc.setMember(bob.publicIdentity, { group: team, role: "member" });
  doc.stage({ type: "note.put", data: { id: "1", title: "hello browser" } }, { group: team });
  await doc.commit();
  const bobs = await openDocument(doc.export(), { model: notes, identity: bob });
  check(bobs.state(team).notes["1"] === "hello browser", "Concord");

  const output = document.createElement("output");
  output.id = "concord-proof";
  output.textContent = `Concord: ${bobs.state(team).notes["1"]}`;
  document.body.append(output);
}
