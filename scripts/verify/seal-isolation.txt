import { isBuiltin, registerHooks } from "node:module";
import assert from "node:assert/strict";
import { readFileSync } from "node:fs";
const fixed = JSON.parse(readFileSync("seal-proof.v2.json","utf8"));
registerHooks({resolve(specifier,context,next){
  if(isBuiltin(specifier) || specifier.startsWith("@ternent/concord") || specifier.includes("ternent-utils") || specifier.startsWith("@ternent/rage"))throw new Error(`Forbidden library dependency: ${specifier}`);
  return next(specifier,context);
}});
for(const path of ["@ternent/seal","@ternent/seal/proof","@ternent/seal/artifact","@ternent/seal/manifest","@ternent/seal/crypto","@ternent/seal/errors"])await import(path);
const {verifySealProofSignature}=await import("@ternent/seal/proof");
assert.equal((await verifySealProofSignature(fixed.proof)).ok,true);
console.log("Seal library imports/verifies with Node builtins, Concord, Utils and Rage imports blocked");
